--- no-opt-inline-constructor-in-a-closure-called-three-times + --no-opt
private class V2 {
	public var x:Float;
	public var y:Float;
	inline public function new(x, y) {
		this.x = x;
		this.y = y;
	}
}
class P {
	inline static function getPos() : V2 {
		return new V2(0, 1);
	}
	inline static public function iteration(func) {
		func();
		func();
		func();
	}
	public static function run() {
		var r = "";
		iteration(function() {
			var wpos = getPos();
			r += '(${wpos.x},${wpos.y})';
		});
		return r;
	}
	static function main() { trace(run()); }
}
